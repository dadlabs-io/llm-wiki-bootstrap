def run_file(name, env=None):
    p = subprocess.run([PY, f"tests/scripts/test_{name}.py"], cwd=SCRATCH, capture_output=True, text=True,
                       encoding="utf-8", errors="replace", timeout=300,
                       env=dict(os.environ, PYTHONIOENCODING="utf-8"))
    lines = p.stdout.strip().splitlines() or [""]
    tail = lines[-1] if lines[-1] else (p.stderr.strip().splitlines() or [""])[-1]
    ok = p.returncode == 0 and re.match(r"^(\d+)/\1 passed", lines[-1]) is not None
    return ok, tail


def line_of(text, old):
    return text[: text.index(old)].count("\n") + 1


# optional 3rd argument: ids to run (a run that stopped part way is resumed with the rest; its result file is merged by hand)
only = set(sys.argv[3].split(",")) if len(sys.argv) > 3 else None

# baseline: every file passes unmutated
base = {}
for f in FILES:
    ok, tail = run_file(f)
    base[f] = tail
    print(f"baseline {f}: {tail}", flush=True)
    assert ok, f"unmutated {f} is not green: {tail}"

results = []
for mid, fname, old, new in M:
    if only and mid not in only:
        continue
    target = SCRATCH / "scripts" / fname
    orig = (Path(r"C:\github.com\llm-wiki-bootstrap\scripts") / fname).read_text(encoding="utf-8")
    assert orig.count(old) == 1, f"{mid}: old text occurs {orig.count(old)} times in {fname}: {old!r}"
    target.write_text(orig.replace(old, new), encoding="utf-8", newline="\n")
    ran, killed_by, tail = [], None, ""
    t0 = time.time()
    try:
        for f in FILES:
            ran.append(f)
            ok, tail = run_file(f)
            if not ok:
                killed_by = f
                break
            # survivors of the four main files go on to the extra files; a mutant already past MAIN
            # is still run against EXTRA so a kill there is recorded
    except subprocess.TimeoutExpired:
        killed_by, tail = f, "timeout"
    finally:
        shutil.copyfile(Path(r"C:\github.com\llm-wiki-bootstrap\scripts") / fname, target)
    r = {"id": mid, "file": f"scripts/{fname}", "line": line_of(orig, old),
         "change": f"{old.strip()}  ->  {new.strip()}", "test_files_run": ran,
         "killed_by": killed_by, "last_line": tail, "killed": killed_by is not None,
         "baseline_of_killer": base.get(killed_by)}
    results.append(r)
    print(f"{mid} {fname}:{r['line']} {'killed by ' + killed_by + ' (' + tail[:60] + ')' if killed_by else 'SURVIVED'}"
          f"  [{time.time() - t0:.0f}s]", flush=True)
    OUT.write_text(json.dumps(results, indent=1), encoding="utf-8")

s = sum(not r["killed"] for r in results)
print(f"total {len(results)}, killed {len(results) - s}, survived {s}")
