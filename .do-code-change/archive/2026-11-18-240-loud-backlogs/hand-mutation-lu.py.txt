def plant(text, orig, new):
    n = text.count(orig)
    assert n == 1, f"matched {n} places: {orig[:80]!r}"
    i = text.index(orig)
    return text[:i] + new + text[i + len(orig):], text.count("\n", 0, i) + 1


def check():
    texts = {f: (REPO / f).read_text(encoding="utf-8") for f in (C, T)}
    bad = 0
    for k, (f, orig, new) in enumerate(MUTANTS):
        try:
            assert orig != new
            plant(texts[f], orig, new)
        except AssertionError as e:
            bad += 1
            print(f"M{k:03d} {f}: {e}")
    print(f"{len(MUTANTS)} mutants, {bad} without exactly one match")
    return bad


def build_base():
    base = SCRATCH / "base"
    if base.exists():
        shutil.rmtree(base)
    files = subprocess.run(["git", "-C", str(REPO), "ls-files", "-z"], capture_output=True, text=True,
                           encoding="utf-8").stdout.split("\0")
    for f in files:
        if not f or f.startswith((".do-code-change/", "archive/", "graphify-out/")):
            continue
        src = REPO / f
        if src.is_file():
            (base / f).parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(src, base / f)
    return base


def run_one(k, base, timeout):
    f, orig, new = MUTANTS[k]
    text = (REPO / f).read_text(encoding="utf-8")
    mutated, line = plant(text, orig, new)
    work = SCRATCH / f"m{k:03d}"
    if work.exists():
        shutil.rmtree(work, ignore_errors=True)
    shutil.copytree(base, work)
    (work / f).write_text(mutated, encoding="utf-8")
    rec = {"id": f"M{k:03d}", "file": f, "line": line, "original": orig.strip("\n"), "mutant": new.strip("\n"),
           "verdict": "survived", "killing_test": None, "detail": ""}
    t0 = time.time()
    for tname in (TESTS_T if f == T else TESTS_C):
        try:
            p = subprocess.run(["uv", "run", "--project", str(REPO), "python", str(work / "tests" / "scripts" / f"{tname}.py")],
                               cwd=work, capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=timeout)
        except subprocess.TimeoutExpired:
            rec.update(verdict="timeout", killing_test=f"{tname}.py", detail=f"timeout {timeout}s")
            break
        out = p.stdout + p.stderr
        if p.returncode != 0:
            syntax = "SyntaxError" in out or "IndentationError" in out
            tail = [l for l in out.strip().splitlines() if l.strip()][-1:] or [""]
            rec.update(verdict="syntax-error" if syntax else "killed", killing_test=f"{tname}.py", detail=tail[0][:200])
            break
        rec["detail"] = ([l for l in p.stdout.strip().splitlines() if l.strip()][-1:] or [""])[0][:120]
    rec["secs"] = round(time.time() - t0)
    shutil.rmtree(work, ignore_errors=True)
    return rec


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--check", action="store_true")
    ap.add_argument("--run")
    ap.add_argument("--jobs", type=int, default=4)
    ap.add_argument("--only")
    ap.add_argument("--timeout", type=int, default=400)
    a = ap.parse_args()
    if a.check or not a.run:
        sys.exit(1 if check() else 0)
    if check():
        sys.exit(1)
    SCRATCH.mkdir(parents=True, exist_ok=True)
    base = build_base()
    ids = [int(x[1:]) for x in a.only.split(",")] if a.only else list(range(len(MUTANTS)))
    results, out = {}, Path(a.run)
    with ThreadPoolExecutor(a.jobs) as ex:
        futs = {ex.submit(run_one, k, base, a.timeout): k for k in ids}
        for fut in futs:
            pass
        from concurrent.futures import as_completed
        for fut in as_completed(futs):
            r = fut.result()
            results[r["id"]] = r
            print(f"[{len(results)}/{len(ids)}] {r['id']} {r['file'].split('/')[-1]}:{r['line']} {r['verdict'].upper()} "
                  f"{r['killing_test'] or ''} ({r['secs']}s)  {r['mutant'][:70]!r}", flush=True)
            out.write_text(json.dumps(sorted(results.values(), key=lambda r: r["id"]), indent=1), encoding="utf-8")
    vs = [r["verdict"] for r in results.values()]
    print({v: vs.count(v) for v in sorted(set(vs))}, "total", len(vs))


main()
